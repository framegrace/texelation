func TestToggleZoom_BroadcastsStateUpdate(t *testing.T) {
    d := newTestDesktop(t) // existing helper or equivalent
    pane := d.activeWorkspace.tree.ActiveLeaf.Pane

    var got []StatePayload
    d.dispatcher.Subscribe(EventStateUpdate, func(e Event) {
        if p, ok := e.Payload.(StatePayload); ok {
            got = append(got, p)
        }
    })

    d.toggleZoom() // or the public path that calls it

    if len(got) == 0 {
        t.Fatalf("expected EventStateUpdate after zoom-in; got none")
    }
    last := got[len(got)-1]
    if !last.Zoomed {
        t.Fatalf("expected Zoomed=true after zoom-in, got %+v", last)
    }
    if last.ZoomedPaneID != pane.ID() {
        t.Fatalf("expected ZoomedPaneID=%x, got %x", pane.ID(), last.ZoomedPaneID)
    }

    before := len(got)
    d.toggleZoom()
    if len(got) <= before {
        t.Fatalf("expected EventStateUpdate after zoom-out; got none")
    }
    if got[len(got)-1].Zoomed {
        t.Fatalf("expected Zoomed=false after zoom-out, got %+v", got[len(got)-1])
    }
}

- [ ] **Step 1: Add the helper**

Above (or below) `fullRender`, add:

```go
// partitionPanes splits the input panes into the normal partition (drawn
// first, with workspace effects applied) and the floating partition (drawn
// on top, modals etc.). The zoomed pane, if any, is forced into the normal
// partition regardless of its ZOrder so floating modals can paint above it.
func partitionPanes(panes []*client.PaneState, zoomed bool, zoomedPane [16]byte) (normal, floating []*client.PaneState) {
    const floatingZOrder = 100
    for _, pane := range panes {
        if pane == nil {
            continue
        }
        switch {
        case zoomed && pane.ID == zoomedPane:
            normal = append(normal, pane)
        case pane.ZOrder >= floatingZOrder:
            floating = append(floating, pane)
        default:
            normal = append(normal, pane)
        }
    }
    return normal, floating
}
```

Replace the in-line loop with:

func TestPartitionPanes_ZoomedPaneStaysInNormal(t *testing.T) {
    zoomedID := [16]byte{0xAA}
    helpID := [16]byte{0xBB}
    splitID := [16]byte{0xCC}

    panes := []*client.PaneState{
        {ID: splitID, ZOrder: 0},
        {ID: helpID, ZOrder: 100},
        {ID: zoomedID, ZOrder: 1000},
    }

    normal, floating := partitionPanes(panes, true, zoomedID)

    var hasZoomInNormal bool
    for _, p := range normal {
        if p.ID == zoomedID {
            hasZoomInNormal = true
        }
    }
    if !hasZoomInNormal {
        t.Fatalf("zoomed pane should be in normal partition; got normal=%v", normal)
    }

    var hasHelpInFloating bool
    for _, p := range floating {
        if p.ID == helpID {
            hasHelpInFloating = true
        }
    }
    if !hasHelpInFloating {
        t.Fatalf("help dialog should be in floating partition; got floating=%v", floating)
    }

    for _, p := range floating {
        if p.ID == zoomedID {
            t.Fatalf("zoomed pane must NOT be in floating partition")
        }
    }
}

        {ID: helpID, ZOrder: 100},
    }

    normal, floating := partitionPanes(panes, false, [16]byte{})

    if len(normal) != 1 || normal[0].ID != splitID {
        t.Fatalf("split pane should be in normal; got %v", normal)
    }
    if len(floating) != 1 || floating[0].ID != helpID {
        t.Fatalf("help dialog should be in floating; got %v", floating)
    }
}
